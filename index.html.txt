<script>

/* ===== DATA ===== */
const days = [
 ["Chest","Core"],
 ["Run"],
 ["Legs"],
 ["Rest"],
 ["Run","Core"],
 ["Run"],
 ["Rest"]
];

/* ===== STATE ===== */
let done = JSON.parse(localStorage.getItem("done")||"{}");
let streak = parseInt(localStorage.getItem("streak")||"0");
let lastDay = localStorage.getItem("lastDay")||"";

/* ===== STREAK SYSTEM ===== */
function updateStreak(){
 const today = new Date().toDateString();

 if(Object.values(done).some(v=>v)){
   if(lastDay !== today){
     streak++;
     localStorage.setItem("streak",streak);
     localStorage.setItem("lastDay",today);
   }
 }
 document.getElementById("streak").innerText = "🔥 Streak: "+streak+" days";
}

/* ===== RENDER ===== */
function render(){
 const app=document.getElementById("app");
 app.innerHTML = days.map((d,i)=>`
   <div class="card">
     <b>Day ${i+1}</b><br>
     ${d.map((s,si)=>{
       const key=i+"-"+si;
       return `<div class="${done[key]?"done":""}">
         ${s}
         <button class="btn" onclick="toggle(${i},${si})">
           ${done[key]?"✓":"Done"}
         </button>
       </div>`;
     }).join("")}
   </div>
 `).join("");

 updateStreak();
}

/* ===== ACTION ===== */
function toggle(i,si){
 const key=i+"-"+si;
 done[key]=!done[key];
 localStorage.setItem("done",JSON.stringify(done));
 render();
}

/* ===== PWA INSTALL HINT ===== */
if(window.matchMedia('(display-mode: standalone)').matches){
 console.log("Running as app");
}

/* ===== INIT ===== */
render();

</script>
