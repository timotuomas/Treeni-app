// Plan state
let currentPhase = 0;
const expanded = {};
const done = {};
const DAYS = ["Mon","Tue","Wed","Thu","Fri","Sat","Sun"];
const todayShort = DAYS[(new Date().getDay()+6)%7];

// Library state
let activeCat = "all";
const exState = {};

function catColor(cat) { return {abs:"#4ade80",chest:"#60a5fa",legs:"#f59e0b"}[cat]||"#a78bfa"; }

/* ── PLAN ── */
function renderPlan() {
  const p = phases[currentPhase];
  document.getElementById("statsRow").innerHTML = [
    {label:"Active days",value:"6"},
    {label:"Min / week",value:phases[currentPhase].days.reduce((a,d)=>a+d.sessions.reduce((b,s)=>b+(s.duration||0),0),0)},
    {label:"Runs",value:p.days.flatMap(d=>d.sessions.filter(s=>s.type==="run")).length+"×4km"},
    {label:"Sessions",value:p.days.flatMap(d=>d.sessions.filter(s=>s.type!=="rest")).length},
  ].map(s=>`<div class="stat-box"><div class="stat-val">${s.value}</div><div class="stat-lbl">${s.label}</div></div>`).join("");

  document.getElementById("phaseTabs").innerHTML = phases.map((ph,i)=>
    `<button class="phase-tab ${i===currentPhase?"active":""}" onclick="setPhase(${i})">
      <div class="tab-label">${ph.label}</div><div class="tab-weeks">${ph.weeks}</div>
    </button>`).join("");

  document.getElementById("phaseDesc").textContent = p.description;

  document.getElementById("legend").innerHTML = Object.entries(TYPE_STYLES).map(([k,ts])=>
    `<div class="legend-item" style="color:${ts.color};background:${ts.bg};border-color:${ts.color}25">
      <span>${ts.icon}</span><span>${ts.label}</span>
    </div>`).join("");

  document.getElementById("daysContainer").innerHTML = p.days.map((d,di)=>renderDay(d,di)).join("");

  document.getElementById("loadRows").innerHTML = [
    {label:"Chest sessions",type:"chest",color:"#60a5fa"},
    {label:"Core sessions",type:"abs",color:"#4ade80"},
    {label:"Legs sessions",type:"legs",color:"#f59e0b"},
    {label:"Runs (4 km ea)",type:"run",color:"#f472b6"},
    {label:"Rest days",type:"_rest",color:"#334155"},
  ].map((row,i)=>{
    const count = row.type==="_rest"
      ? p.days.filter(d=>d.sessions.every(s=>s.type==="rest")).length
      : p.days.flatMap(d=>d.sessions.filter(s=>s.type===row.type)).length;
    const dots = count>0
      ? Array.from({length:count},()=>`<div class="load-dot" style="background:${row.color}30;border:1px solid ${row.color}60"></div>`).join("")
      : `<span style="color:#2a2a3a;font-size:12px">—</span>`;
    return `<div class="load-row" ${i===4?'style="border-bottom:none"':""}>
      <span class="load-lbl">${row.label}</span><div class="load-dots">${dots}</div>
    </div>`;
  }).join("");
}

function renderDay(d,di) {
  const isRest = d.sessions.every(s=>s.type==="rest");
  const isToday = d.day===todayShort;
  const isExp = expanded[di];
  const totalM = d.sessions.reduce((a,s)=>a+(s.duration||0),0);
  const allDone = d.sessions.length>0 && d.sessions.every((_,si)=>done[`${currentPhase}-${di}-${si}`]);
  const pills = d.sessions.map((s,si)=>{
    const ts=TYPE_STYLES[s.type]||TYPE_STYLES.rest;
    const isDone=done[`${currentPhase}-${di}-${si}`];
    const t=s.type==="run"?`Run ${s.duration}min`:s.name.split(" ")[0]+(s.duration?` ${s.duration}m`:"");
    return `<div class="pill ${isDone?"done":""}" style="${isDone?"":"color:"+ts.color+";background:"+ts.bg+";border-color:"+ts.color+"30"}">
      <span>${isDone?"✓":ts.icon}</span><span>${t}</span></div>`;
  }).join("");
  const body = isExp ? `<div class="day-body">${d.sessions.map((s,si)=>{
    const ts=TYPE_STYLES[s.type]||TYPE_STYLES.rest;
    const isDone=done[`${currentPhase}-${di}-${si}`];
    return `<div class="session-block">
      <div class="session-top">
        <div class="session-title">
          <span style="font-size:16px">${ts.icon}</span>
          <span class="session-name" style="color:${ts.color}">${s.name}</span>
          ${s.duration?`<span class="session-dur">${s.duration} min</span>`:""}
        </div>
        <button class="mark-btn ${isDone?"done":""}" onclick="toggleDone(${di},${si},event)">${isDone?"✓ done":"mark done"}</button>
      </div>
      <div class="details">${s.details.map(det=>`<div class="detail-line" style="border-left-color:${ts.color}25">${det}</div>`).join("")}</div>
    </div>`;
  }).join("")}</div>` : "";
  return `<div class="day-card ${isRest?"rest-day":""} ${allDone?"all-done":""} ${isToday?"today":""}">
    <div class="day-header" onclick="toggleExpand(${di})">
      <div class="day-label-wrap">
        <div class="day-name ${isToday?"today-name":""}">${d.day}</div>
        ${isToday?"<div class='today-dot'></div>":""}
      </div>
      <div class="session-pills">${pills}</div>
      ${totalM>0?`<div class="day-total">${totalM}min</div>`:""}
      ${!isRest?`<div class="expand-icon">${isExp?"▲":"▼"}</div>`:""}
    </div>${body}</div>`;
}

function setPhase(i){currentPhase=i;renderPlan();}
function toggleExpand(di){expanded[di]=!expanded[di];renderPlan();}
function toggleDone(di,si,e){e.stopPropagation();const k=`${currentPhase}-${di}-${si}`;done[k]=!done[k];renderPlan();}

/* ── LIBRARY ── */
function renderLibrary() {
  document.getElementById("catFilters").innerHTML = CATS.map(c=>
    `<button class="cat-pill ${activeCat===c.id?"active":""}" style="--c:${c.color}" onclick="setCat('${c.id}')">${c.label}</button>`
  ).join("");
  const filtered = activeCat==="all" ? exercises : exercises.filter(e=>e.cat===activeCat);
  document.getElementById("exList").innerHTML = filtered.map(ex=>{
    const color=catColor(ex.cat);
    const isOpen=exState[ex.name];
    const catLabel={abs:"Abs & Core",chest:"Chest",legs:"Legs & Glutes"}[ex.cat]||ex.cat;
    const safe=ex.name.replace(/'/g,"\\'");
    return `<div class="ex-card">
      <div class="ex-card-header" onclick="toggleEx('${safe}')">
        <div class="ex-icon-wrap" style="background:${color}10;border-color:${color}25"><span>${ex.icon}</span></div>
        <div class="ex-card-info">
          <div class="ex-card-name" style="color:${color}">${ex.name}</div>
          <div class="ex-card-meta">${catLabel} · ${ex.phase}</div>
        </div>
        <div class="ex-chevron">${isOpen?"▲":"▼"}</div>
      </div>
      ${isOpen?`<div class="ex-card-body">
        <div class="ex-sets-badge" style="border-color:${color}20"><span style="color:${color}">📊</span><span>${ex.sets}</span></div>
        <div>
          <div class="ex-section-label" style="color:${color}">Why this exercise</div>
          <div class="ex-text">${ex.why}</div>
        </div>
        <div>
          <div class="ex-section-label" style="color:${color}">How to do it</div>
          <div class="ex-how-box" style="border-color:${color}15"><div class="ex-how-text">${ex.how}</div></div>
        </div>
      </div>`:""}
    </div>`;
  }).join("");
}

function setCat(cat){activeCat=cat;Object.keys(exState).forEach(k=>delete exState[k]);renderLibrary();}
function toggleEx(name){exState[name]=!exState[name];renderLibrary();}

  document.getElementById("pagePlan").classList.toggle("active",page==="plan");
  document.getElementById("pageExercises").classList.toggle("active",page==="exercises");
  document.getElementById("navPlan").classList.toggle("active",page==="plan");
  document.getElementById("navExercises").classList.toggle("active",page==="exercises");
  if(page==="exercises") renderLibrary();
}
